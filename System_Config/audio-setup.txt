===========================================================
 WIREPLUMBER CONFIG: ~/.config/wireplumber/wireplumber.conf.d/bluetooth-a2dp-autoconnect.conf
===========================================================

## Auto-connect A2DP playback/capture profiles on Bluetooth devices.
## This helps speakers and receivers expose their audio profiles without
## requiring manual PipeWire/WirePlumber recovery.

monitor.bluez.rules = [
  {
    matches = [
      {
        device.name = "~bluez_card.*"
      }
    ]
    actions = {
      update-props = {
        bluez5.auto-connect = [ a2dp_sink a2dp_source ]
      }
    }
  }
]

===========================================================
 QUICK SETUP ON THE OTHER SYSTEM
===========================================================

1. Install packages with the oneliner above
2. Create the wireplumber config:
   mkdir -p ~/.config/wireplumber/wireplumber.conf.d
   (then place the bluetooth-a2dp-autoconnect.conf in that directory)
3. Enable and start:
   systemctl --user enable --now pipewire pipewire-pulse wireplumber
